cmake_minimum_required(VERSION 3.24)
project(RetroMediaLab VERSION 0.1.0 LANGUAGES CXX)
if(MSVC AND DEFINED RETROMEDIALAB_MSVC_INCLUDE_PREFIX)
  set(CMAKE_CL_SHOWINCLUDES_PREFIX "${RETROMEDIALAB_MSVC_INCLUDE_PREFIX}")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets)
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8 /EHsc)
endif()
add_subdirectory(src/Translations)
add_subdirectory(src/Platform)
add_subdirectory(src/App)
install(DIRECTORY DESTINATION Modules)
install(FILES src/App/Assets/app-icon.ico DESTINATION Assets)
